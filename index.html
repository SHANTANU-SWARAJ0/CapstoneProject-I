import re
import math

def calculate_entropy(password):
    """Estimate password entropy in bits."""
    charset = 0
    if re.search(r'[a-z]', password): charset += 26
    if re.search(r'[A-Z]', password): charset += 26
    if re.search(r'[0-9]', password): charset += 10
    if re.search(r'[^a-zA-Z0-9]', password): charset += 32
    return round(len(password) * math.log2(charset), 2) if charset else 0

def check_password(password):
    """Run all security checks. Returns a result dict."""
    score  = 0
    issues = []
    tips   = []

    # Rule 1 — Minimum length (12 chars)
    if len(password) >= 12: score += 20
    else:
        issues.append("Too short — need 12+ characters")
        tips.append("Use at least 12 characters")

    # Rule 2 — Uppercase letters
    if re.search(r'[A-Z]', password): score += 20
    else:
        issues.append("No uppercase letters")
        tips.append("Add uppercase letters (A–Z)")

    # Rule 3 — Digits
    if re.search(r'[0-9]', password): score += 20
    else:
        issues.append("No digits found")
        tips.append("Mix in numbers (0–9)")

    # Rule 4 — Special characters
    if re.search(r'[^a-zA-Z0-9]', password): score += 20
    else:
        issues.append("No special characters")
        tips.append("Add symbols: !@#$%^&*()")

    # Rule 5 — No common patterns
    common = ['123456', 'password', 'qwerty', 'abc123', 'admin']
    if not any(p in password.lower() for p in common):
        score += 20
    else:
        issues.append("Contains a common pattern")
        tips.append("Avoid predictable patterns")

    entropy = calculate_entropy(password)

    if   score >= 100: level = "Very Strong"
    elif score >= 80:  level = "Strong"
    elif score >= 60:  level = "Medium"
    elif score >= 40:  level = "Weak"
    else:               level = "Very Weak"

    return {"score":score, "level":level, "entropy":entropy, "issues":issues, "tips":tips}
